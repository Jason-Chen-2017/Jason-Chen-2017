🏗️ 部署架构

Kubernetes 部署配置

apiVersion: apps/v1
kind: Deployment
metadata:
  name: research-automation
  namespace: research-automation
spec:
  replicas: 3
  selector:
    matchLabels:
      app: research-automation
  template:
    spec:
      containers:
      - name: app
        image: registry.example.com/research-automation-app:latest
        ports:
        - containerPort: 8080
        resources:
          requests:
            cpu: "500m"
            memory: "512Mi"
          limits:
            cpu: "2000m"
            memory: "2Gi"
        livenessProbe:
          httpGet:
            path: /health/live
            port: 8080
        readinessProbe:
          httpGet:
            path: /health/ready
            port: 8080
